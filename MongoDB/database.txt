// MongoDB Database: ctpga_management

// Crear la base de datos
use ctpga_management

// Crear colecciones
db.createCollection("instructors")
db.createCollection("actividades_formacion")
db.createCollection("reportes")
db.createCollection("eventos")
db.createCollection("admin")

// Crear índices
// Índices para instructors
db.instructors.createIndex({"no_documento_identidad": 1}, {unique: true})
db.instructors.createIndex({"correo": 1}, {unique: true})
db.instructors.createIndex({"contraseña": 1}, {unique: true})

// Índices para actividades_formacion
db.actividades_formacion.createIndex({"numero_ficha": 1})
db.actividades_formacion.createIndex({"estado": 1})
db.actividades_formacion.createIndex({"horario.fecha_inicio": 1, "horario.fecha_fin": 1})

// Índices para reportes
db.reportes.createIndex({"periodo.mes": 1, "periodo.año": 1})
db.reportes.createIndex({"instructor.instructor_id": 1})
db.reportes.createIndex({"estado": 1})

// Índices para eventos
db.eventos.createIndex({"fecha_evento": 1})
db.eventos.createIndex({"estado": 1})

// Índices para admin
db.admin.createIndex({"usuario": 1}, {unique: true})
db.admin.createIndex({"correo": 1}, {unique: true})
db.admin.createIndex({"nivel_acceso": 1})
